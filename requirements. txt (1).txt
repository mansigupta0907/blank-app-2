import io
from pathlib import Path
import zipfile
import streamlit as st

st.title("Dashboard Image Downloader")

# 1. Define image paths (relative to app directory)
image_paths = [
    "dashboards/P1_Executive.png",
    "dashboards/P2_Shipping.png",
    "dashboards/P3_Product.png",
    "dashboards/P4_Regional.png",
    "dashboards/P5_Customer.png",
    "dashboards/P6_Profit.png",
]


# 2. Function to create a ZIP file in memory
def create_zip(paths):
    zip_buffer = io.BytesIO()
    with zipfile.ZipFile(zip_buffer, "w", zipfile.ZIP_DEFLATED) as zip_file:
        for path_str in paths:
            file_path = Path(path_str)
            if file_path.exists():
                # Add file to zip using its filename
                zip_file.write(file_path, arcname=file_path.name)
    zip_buffer.seek(0)
    return zip_buffer


# 3. Create the ZIP buffer
zip_data = create_zip(image_paths)

# 4. Streamlit Download Button
st.download_button(
    label="📦 Download All Images (ZIP)",
    data=zip_data,
    file_name="dashboards.zip",
    mime="application/zip",
)
